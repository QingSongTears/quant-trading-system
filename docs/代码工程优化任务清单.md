```python
"""SQLAlchemy Engine 单例 — 全项目唯一数据库连接池"""
from sqlalchemy import create_engine, Engine
from src.config import get_config, get_db_url

_engine: Engine | None = None

def get_engine() -> Engine:
    global _engine
    if _engine is None:
        config = get_config()
        db_url = get_db_url(config)
        _engine = create_engine(
            db_url,
            echo=config.get("database", {}).get("echo", False),
            pool_size=5,
            max_overflow=10,
        )
    return _engine

def reset_engine():
    """仅测试用：重置 engine（关闭旧连接）"""
    global _engine
    if _engine is not None:
        _engine.dispose()
    _engine = None
```

### E7. 统一类型注解风格

**问题**: `Optional[X]` / `X | None` 混用，`from __future__ import annotations` 不统一。

#### 对所有 `src/` 下 .py 文件执行统一替换

```python
# === BEFORE ===
from typing import Dict, List, Optional, Tuple, Any

def foo(x: Optional[str]) -> List[Dict[str, Any]]:
    ...

# === AFTER ===
from __future__ import annotations
from typing import Any

def foo(x: str | None) -> list[dict[str, Any]]:
    ...
```

**具体改动清单**:

| 原类型 | 替换为 |
|--------|--------|
| `Optional[X]` | `X \| None` |
| `List[X]` | `list[X]` |
| `Dict[K,V]` | `dict[K,V]` |
| `Tuple[X,Y]` | `tuple[X,Y]` |
| `from typing import ...` | 仅保留 `Any`, `Callable`, `TypeVar` 等无法简化的 |

### E9. 夏普计算修正

**问题**: `src/backtest/engine.py` `_calc_sharpe` 基于底层资产价格而非策略净值计算。

#### 修改 `src/backtest/engine.py` 第338-343行

```python
# === BEFORE ===
def _calc_sharpe(self, df: pd.DataFrame, initial: float, final: float, days: int) -> float:
    """计算夏普比率"""
    if days <= 1:
        return 0
    daily_returns = df["Close"].pct_change().dropna()
    if len(daily_returns) == 0:
        return 0
    excess = daily_returns.mean() * 250 - self.risk_free_rate
    vol = daily_returns.std() * np.sqrt(250)
    return excess / vol if vol > 0 else 0

# === AFTER ===
def _calc_sharpe(self, stats: dict, initial: float, days: int) -> float:
    """计算夏普比率 — 基于策略净值曲线"""
    if days <= 1:
        return 0
    equity = stats.get("_equity_curve", None)
    if equity is None:
        return 0
    # backtesting 返回 DataFrame 时取第一列 (Equity)
    if hasattr(equity, 'columns'):
        equity_series = pd.Series(equity.iloc[:, 0].values)
    elif hasattr(equity, 'values'):
        equity_series = pd.Series(equity.values)
    else:
        return 0
    daily_returns = equity_series.pct_change().dropna()
    if len(daily_returns) == 0:
        return 0
    excess = daily_returns.mean() * 250 - self.risk_free_rate
    vol = daily_returns.std() * np.sqrt(250)
    return excess / vol if vol > 0 else 0
```

#### 13.5 数字滚动动画（Counting Animation）

**建议**: 指标卡片加载时使用数字滚动效果（`countUp` 式动画）。

**实现** — 纯 CSS + JS，无需额外依赖:

```javascript
// 数字递增动画
function animateValue(el, start, end, duration = 1000) {
  const range = end - start;
  const startTime = performance.now();
  const step = (now) => {
    const elapsed = now - startTime;
    const progress = Math.min(elapsed / duration, 1);
    // ease-out
    const value = start + range * (1 - Math.pow(1 - progress, 3));
    el.textContent = el.dataset.prefix || '' + value.toFixed(el.dataset.decimals || 0) + (el.dataset.suffix || '');
    if (progress < 1) requestAnimationFrame(step);
  };
  requestAnimationFrame(step);
}

// 用法:
// <span data-prefix="+" data-suffix="%" data-decimals="2" data-target="15.82">0</span>
document.querySelectorAll('[data-target]').forEach(el => {
  animateValue(el, 0, parseFloat(el.dataset.target), 1200);
});
```

#### 13.9 Toast 通知（替代 alert）

**现状**: 错误提示靠 `alert()` 或页面直接显示红色文字，体验生硬。

**建议**: 用 Bootstrap 5 自带的 Toast 组件做非阻塞通知。

```javascript
// 全局 Toast 工具函数（放入 base.html <script>）
function showToast(message, type = 'info') {
  const colors = { success: 'bg-success', error: 'bg-danger', warning: 'bg-warning', info: 'bg-info' };
  const icons = { success: 'check-circle', error: 'exclamation-triangle', warning: 'exclamation-circle', info: 'info-circle' };
  
  const container = document.getElementById('toast-container') || (() => {
    const c = document.createElement('div');
    c.id = 'toast-container';
    c.className = 'toast-container position-fixed bottom-0 end-0 p-3';
    c.style.zIndex = '9999';
    document.body.appendChild(c);
    return c;
  })();
  
  const id = 'toast-' + Date.now();
  container.innerHTML += `
    <div id="${id}" class="toast align-items-center text-white ${colors[type]} border-0" role="alert">
      <div class="d-flex">
        <div class="toast-body"><i class="bi bi-${icons[type]} me-2"></i>${message}</div>
        <button class="btn-close btn-close-white me-2 m-auto" data-bs-dismiss="toast"></button>
      </div>
    </div>`;
  new bootstrap.Toast(document.getElementById(id), { delay: 4000 }).show();
}

// 全局替换: alert('回测失败') → showToast('回测失败', 'error')
```

**批量执行提示**: E7 可以用以下命令批量改：
```bash
# 添加 from __future__ import annotations（跳过已有文件）
rg -L "^from __future__ import annotations" src/ --type py | xargs sed -i '1s/^/from __future__ import annotations\n/'

# 替换 Optional[X] → X | None
find src/ -name "*.py" -exec sed -i 's/Optional\[\(.*\)\]/\1 | None/g' {} +

# 替换 List[X] → list[X]
find src/ -name "*.py" -exec sed -i 's/List\[\(.*\)\]/list[\1]/g' {} +
```
